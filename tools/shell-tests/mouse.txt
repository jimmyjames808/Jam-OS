# The mouse, end to end: run by tools/mouse-test.sh (QEMU_USB = a usb-kbd
# on xhci port 2 and a usb-mouse on port 3; the command line "shell", at
# OVMF's 1280x800). QEMU's monitor moves the mouse and presses its buttons
# (`mouse_move dx dy [dz]`, `mouse_button <bits>`: 1 left, 2 right;
# serial-feed's `pointer x y` puts it somewhere exact); the reports go
# usb-bus -> drv/hid -> the compositor, which moves its pointer and sends
# the window under it wl_pointer's events.
#
# The client that asks is `wltest --mouse` (user/tests/wltest/fun.c),
# which goes through libfun as an app does: its window, and libfun's
# mouse (user/apps/fun/mouse.c). It says every edge on the console:
# "wltest: mouse: left down at X,Y", "... left click at X,Y" (or "drag
# from X,Y to X,Y"), "... chord at X,Y", "... wheel +1 at X,Y", in its
# window's pixels. The windows tile (the default): wltest's opens beside
# the first terminal, in the right half (x 643..1274, y 46..794), and
# takes the keys; the clicks are in it, at (960, 420) and around.
wait 120 Jam OS shell
wait {prompt}
# "report mouse ready", or with hidboot "boot mouse ready" (mouse-test.sh
# checks which).
seen 30 mouse ready
# The shell never asked for the mouse: moving it, clicking in the
# terminal (a selection, the terminal's own) and turning the wheel sends
# it nothing (its line editor would misread any message).
send help
wait Commands:
wait {prompt}
pointer 300 400
monitor mouse_button 1
sleep 0.2
monitor mouse_button 2
sleep 0.2
monitor mouse_button 0
monitor mouse_move 0 0 1
monitor mouse_move 0 0 -1
sleep 0.7
shot mouse-shell
send echo mouse-$HOME
wait mouse-/
wait {prompt}
# An app that didn't ask for the mouse (`wltest --mouse nomouse`) reads
# its keys undisturbed under a busy mouse: only the two keys typed reach it.
send run wltest --mouse nomouse
wait run: bin/wltest started
wait 30 wltest: mouse: ready
sleep 1
pointer 960 420
monitor mouse_button 1
sleep 0.2
monitor mouse_button 0
monitor mouse_move -9 4 1
type x
sleep 0.5
type q
wait 30 wltest: mouse: 0 clicks (0 left, 0 right, 0 middle), 0 drags, 0 chords, wheel 0 (0 notches), 2 keys, 0 mouse reports
wait run: bin/wltest exited with code 0
wait {prompt}
# wltest asked: a left click where the pointer is put, then a right click
# held for a screenshot, a chord, a drag each way and the wheel, each
# said at the pointer's place in the window.
send run wltest --mouse noaccel
wait run: bin/wltest started
wait 30 wltest: mouse: ready, {ww}x{wh} (a window), pointer at
sleep 1
pointer 960 420
sleep 0.3
monitor mouse_button 1
sleep 0.2
monitor mouse_button 0
wait 10 wltest: mouse: left down at {cx},{cy}
wait 10 wltest: mouse: left click at {cx},{cy}
monitor mouse_button 2
wait 10 wltest: mouse: right down at {cx},{cy}
sleep 0.5
shot mouse-right-held
monitor mouse_button 0
wait 10 wltest: mouse: right click at {cx},{cy}
# A chord: left, then right as well; both let go at once.
monitor mouse_button 1
sleep 0.2
monitor mouse_button 3
wait 10 wltest: mouse: chord at {cx},{cy} (left and right held)
sleep 0.2
monitor mouse_button 0
wait 10 wltest: mouse: left click at {cx},{cy}
wait 10 wltest: mouse: right click at {cx},{cy}
# Drags: left down, moved, up; then the right button back the other way.
# (steps of 6 and 3 counts: the acceleration takes them 1:1, 60 and 30
# pixels in all)

sleep 0.2

monitor mouse_move 6 3
monitor mouse_move 6 3
monitor mouse_move 6 3
monitor mouse_move 6 3
monitor mouse_move 6 3
monitor mouse_move 6 3
monitor mouse_move 6 3
monitor mouse_move 6 3
monitor mouse_move 6 3
sleep 0.3
monitor mouse_button 0
wait 10 wltest: mouse: left drag from {cx},{cy} to {dx},{dy}
monitor mouse_button 2
sleep 0.2
monitor mouse_move -6 -3
monitor mouse_move -6 -3
monitor mouse_move -6 -3
monitor mouse_move -6 -3
monitor mouse_move -6 -3
monitor mouse_move -6 -3
monitor mouse_move -6 -3
monitor mouse_move -6 -3
monitor mouse_move -6 -3
monitor mouse_move -6 -3
sleep 0.3
monitor mouse_button 0
wait 10 wltest: mouse: right drag from {dx},{dy} to {cx},{cy}
# The wheel, a notch each way.
monitor mouse_move 0 0 1
wait 10 wltest: mouse: wheel
sleep 0.2
monitor mouse_move 0 0 -1
wait 10 wltest: mouse: wheel
sleep 0.5
shot mouse-wltest
type q
wait 30 wltest: mouse: 4 clicks (2 left, 2 right, 0 middle), 2 drags, 1 chords, wheel 0 (2 notches), 1 keys,
wait run: bin/wltest exited with code 0
