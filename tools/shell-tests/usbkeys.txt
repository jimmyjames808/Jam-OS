# Typing into the shell on a USB keyboard: QEMU's keys go
# usb-bus -> drv/hid -> console -> shell, the output is checked on COM1.
# Run by tools/usbkeys-test.sh: QEMU_USB = a usb-hub on xhci port 2 with a
# usb-kbd behind it (usb path 6.1: hid-6.1:0) and a usb-mouse on port 3;
# the command line "shell" (a plain boot).
wait 120 Jam OS shell
wait jam>
seen 30 hid-6.1:0: hid 0627:0001 if 0: boot keyboard ready
sleep 0.5
usbkeys help\r
wait Commands:
wait jam>
# kill the keyboard's hid (from the serial side): devmgr restarts it, and
# the keys typed right away (QEMU queues them while nobody polls) still arrive
send kill hid-6.1:0
wait shell: killed process
seen 10 usb 6.1:0 drv/hid restarted
sleep 0.5
usbkeys help\r
wait 10 Commands:
wait jam>
# the console restarts: hid loses its input channel, ends, and devmgr
# starts it again connected to the new console
send kill console
wait 30 Jam OS shell.
seen 30 usb 6.1:0 drv/hid: its console went away: reconnecting
seen 30 usb 6.1:0 drv/hid started again (the console is back)
sleep 1
usbkeys help\r
wait 10 Commands:
wait jam>
# devmgr dies: init kills its job (usb-bus and every hid go
# with it) and starts it again; the new usb-bus resets the controller and
# enumerates, a new hid binds and the keyboard types again; the shell's
# usb command talks to the new devmgr
send kill devmgr
# (init's line goes to the kernel log, the shell's through the console: their
# order on the serial port isn't fixed, hence `seen`)
wait shell: killed process
seen 30 init: devmgr and its drivers are gone
wait 90 hid-6.1:0: hid 0627:0001 if 0: boot keyboard ready
sleep 1
usbkeys help\r
wait 10 Commands:
wait jam>
send usb
wait 30 problem(s)
wait jam>
send reboot -f
wait reboot: resetting
