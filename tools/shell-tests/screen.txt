# The screen on a plain boot, the desktop's (tools/screen-test.sh;
# QEMU_SPLASH=1, so the first terminal's console gets "nolog"): after the
# splash, the shell's banner and prompt in the first terminal's window and
# no kernel log in it; the log there while a command whose output it is
# runs; and the desktop's notices (devmgr's, through /svc/notify): another
# stick plugged in and pulled out (a remount in between says nothing), the
# boot stick pulled out and back. (Before M12's Q14 this was the
# full-screen console's own screen, booted with `nocomp`.)
wait 120 init: bin/splash exited with code 0
seen 60 {prompt}
sleep 2
shot screen-boot

# ---- another stick -----------------------------------------------------------------
monitor device_add usb-storage,id=sa,bus=xhci.0,port=2,drive=aimg
wait 60 compositor: notice {a}: USB stick added:
sleep 1
shot screen-stick
send ls /usb0
wait hello.txt
wait {prompt}
send mount -w /usb0
wait {prompt}
sleep 5
send mount -r /usb0
wait {prompt}
sleep 5
send echo remount-quiet
wait remount-quiet
monitor device_del sa
wait 30 compositor: notice {b}: USB stick removed: /usb0

# ---- the log while a command runs ----------------------------------------------------
send ktest utf8
wait shell: ktest utf8: 5 passed
wait {prompt}
sleep 1
shot screen-ktest
send clear
wait {prompt}

# ---- the boot stick ----------------------------------------------------------------
monitor device_del stick
wait 30 compositor: notice {c}: Jam OS stick removed
monitor device_add usb-storage,id=stick,bus=xhci.0,port=1,drive=usbstick
wait 60 compositor: notice {d}: Jam OS stick is back
sleep 1
shot screen-back
send reboot -f
wait reboot: resetting
