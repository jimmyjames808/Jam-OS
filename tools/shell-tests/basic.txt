# The shell over COM1: tools/qemu-test.sh with QEMU_INPUT=this file and
# the command line "shell" (what a plain boot does), or "shell nocomp".
# Types through the serial port -> serialin -> compositor -> the terminal's
# console -> shell path (under `nocomp`, serialin -> console -> shell) and
# checks what the shell writes back (the console mirrors program output to
# COM1).
wait 120 Jam OS shell
wait jam>
send help
wait restart the machine
wait jam>
sleep 0.5
shot shell-help
# line editing: "helx", backspace, "p" -> help; then the history (up, Enter)
type helx\x7fp\r
wait Commands:
wait jam>
type \e[A\r
wait Commands:
wait jam>
# "hlp", left, left, "e" -> "help"; Home, Delete, End
type hlp\e[D\e[De\r
wait Commands:
wait jam>
type xhelp\e[H\e[3~\e[F\r
wait Commands:
wait jam>
send nosuchcommand
wait nosuchcommand: unknown command
send usb
wait usb:
wait problem(s)
wait jam>
send devices
wait devmgr:
send ps
wait process
wait jam>
send mem
wait MiB free
send log 5
wait jam>
# the console's protocols from a program's (PROGRAM-level) channel: no
# input sources or new channels; the focus stack with keys typed here
send run contest
wait 30 contest: type q now
type q
wait contest: type r now
type r
wait contest: type s now
type s
wait 60 run: bin/contest exited with code 0
wait jam>
# a program writing to the console flat out: the console serves it a
# bounded round at a time, so Ctrl+C (a key) and the shell's own output
# still get through
send run contest flood
wait 30 contest: flood: writing flat out
sleep 2
type \x03
wait 10 ^C: killing bin/contest
wait 10 run: bin/contest was killed
wait jam>
send help
wait Commands:
wait jam>
send run bin/utest
wait 240 run: bin/utest exited with code 0
wait jam>
send ktest m4
wait 300 shell: ktest m4:
wait passed
wait jam>
sleep 0.5
shot shell-ktest
# restarts: the serial source, then the console itself (init restarts it
# and the shell comes back; with the compositor its window goes and comes
# back, under `nocomp` the kernel draws its log meanwhile and serialin
# comes back too: screen-test.sh looks at that screen)
send kill serialin
wait bin/serialin was killed
sleep 1
send help
wait Commands:
wait jam>
send kill console
wait 30 init: bin/console was killed
wait 30 Jam OS shell
wait jam>
sleep 1
send help
wait Commands:
wait jam>
sleep 0.5
shot shell-restarted
send reboot -f
wait reboot: resetting
