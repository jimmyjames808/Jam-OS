# The ported programs (M13 P1): sbase's utilities and Lua, in the boot image's
# usr/bin, run from the shell on real files in /data. The shell has its own
# command for many of the names (cat sort uniq wc head tail grep ls cp mv rm
# mkdir find seq date ...) and those win by name, so sbase's version of them
# is run by its path: `run $S/<name>` (S=/boot/usr/bin); the others (sed tr cut
# xargs od printf paste cmp ...) have no shell command and run by name.
# Every check compares a program's output with a file made by printf (also
# sbase's) using cmp (sbase's): `cmp got want && echo ok-$T-name`, so the
# text waited for (built from $T) never appears in the typed line.
#   QEMU_INPUT=tools/shell-tests/ports.txt tools/qemu-test.sh <dir> ports shell
wait 120 Jam OS shell
wait {prompt}
send T=x
wait {prompt}
send S=/boot/usr/bin
wait {prompt}
send cd /data
wait {prompt}

# ---- by name: the programs the shell has no command for ------------------------------------------
send printf 'pear\napple\nfig\napple\ncherry\nfig\nfig\n' > /data/pt-w; echo st-$T=$?
wait st-x=0
wait {prompt}
send printf 'x %s\n' one two > /data/pt-e; cmp /data/pt-e /data/pt-e && echo ok-$T-printf
wait ok-x-printf
wait {prompt}
send cat /data/pt-e
wait x one
wait x two
wait {prompt}

# ---- sbase's version is found by its path (the shell's own command wins by name) ----------------
