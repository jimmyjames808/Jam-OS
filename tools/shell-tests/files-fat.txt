# The shell's file commands on a real FAT volume: `run utest fat-shell`
# starts a second shell whose /data is the fat service (bin/fat) over a
# RAM disk, freshly formatted (Ctrl+C ends it). Names with spaces and lower
# case, names that differ only after eight characters, forbidden
# characters, copies bigger than a transfer buffer, rm -r, a directory
# bigger than ls shows (it says so), df and sync.
# files.txt has the same commands on the tests' RAM filesystem.
# QEMU_INPUT=this file, command line "shell". As in cmds.txt, the text
# waited for is built from $T, so it never appears in the typed line.
wait 120 Jam OS shell
wait jam>
send run utest fat-shell
# (the outer shell's "started" line, the second shell's banner and fat's
# lines in the kernel log have no fixed order on the serial port: `seen`)
seen 60 run: bin/utest started
seen 60 fat utest: mounted
sleep 1
send T=f
wait jam>
send df | grep /data | grep -c JAMOS-DATA | grep ^1$ && echo ok-$T-df
wait ok-f-df
send cd /data && ls | wc -l | grep ^0$ && echo ok-$T-empty
wait ok-f-empty

# ---- names as typed ---------------------------------------------------------------------
send touch "My Notes.txt" && ls | grep -c "^My Notes.txt$" | grep ^1$ && echo ok-$T-spaces
wait ok-f-spaces
send write notes.txt lower case stays && ls | grep -c ^notes.txt$ | grep ^1$ && echo ok-$T-lower
wait ok-f-lower
send write holiday-photos.txt one-$T; write holiday-plans.txt two-$T
send cat holiday-photos.txt holiday-plans.txt
wait one-f
wait two-f
send ls | grep -c holiday | grep ^2$ && echo ok-$T-aliases
wait ok-f-aliases
send touch "bad:name"; echo st-$T=$?

wait st-f=1
send mkdir -p "Trip 2026/day 1" && cd "Trip 2026/day 1" && pwd | grep "^/data/Trip 2026/day 1$" && echo ok-$T-dirs
wait ok-f-dirs
send cd /data

wait 60 hex-f-done
send cat end1 end2 | sort | uniq | wc -l | grep ^3$ && echo ok-$T-cpbig
wait ok-f-cpbig
send mv "My Notes.txt" "Trip 2026" && ls "Trip 2026" | grep -c "^My Notes.txt$" | grep ^1$ && echo ok-$T-mv
wait ok-f-mv
send mv notes.txt "Notes Again.txt" && cat "Notes Again.txt" | head -1
wait lower case stays
send find /data -name copy
wait /data/Trip 2026/utest copy
send run utest fscat "/data/Notes Again.txt"
wait fscat: lower case stays
wait run: bin/utest exited with code 0
send rm "Trip 2026"; echo st-$T=$?
wait rm: Trip 2026: a directory with entries
wait st-f=1
send rm -r "Trip 2026" end1 end2 && ls | grep -c Trip | grep ^0$ && ls | grep -c ^end | grep ^0$ && echo ok-$T-rmr
wait ok-f-rmr
# ---- cat of a file bigger than any command reads whole (4 MiB): it streams ---------------
# (the pipe keeps the first 4 MiB and says so; head has its line)
send seq 1 400000 | write long.txt; seq 1 400000 | write -a long.txt; ls -l long.txt
wait MiB  long.txt
wait jam>
send cat long.txt | head -1 | grep ^1$ && echo ok-$T-stream
wait 60 ok-f-stream
send rm long.txt
