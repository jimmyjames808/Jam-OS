# fbbench (user/tests/fbbench): run from the shell with a tenth of the
# samples (-q). It opens a full-screen window, measures, closes it and
# prints its lines: each section's lines must be there, none FAILED, and
# the shell must come back; then a usage error. Command line "shell". The
# numbers are QEMU's (TCG): meaningless, the PC gives them. The text
# waited for is built from $T, so it never appears in the typed line.
wait 120 Jam OS shell
wait {prompt}
send T=f
wait {prompt}
send fbbench -q; echo st-$T=$?
wait 60 fbbench: kernels agree with px_over
wait 600 fbbench: a full-screen window
wait 60 fbbench: median and p99 of
seen 60 fbbench: window: libfun present, every piece changed
seen 60 fbbench: window: libfun present, nothing changed
seen 60 fbbench: RAM copy: 16-byte, whole screen's bytes, 1 thread
seen 60 fbbench: blend: argb over xrgb, scalar (px_over), whole screen, 1 thread
seen 60 fbbench: blend: argb over xrgb, SSE2, whole screen, 1 thread
seen 60 fbbench: vmo_read: 4 KiB in one call, 1 thread
seen 60 fbbench: vmo_read: whole screen's bytes in 16-row bands
wait 60 fbbench: done
wait st-f=0
send echo back-$T
wait back-f
send fbbench -x; echo st-$T=$?
wait usage: fbbench
wait st-f=2
send reboot -f
wait reboot: resetting
