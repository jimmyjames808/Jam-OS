# The boot splash (user/apps/splash), booted with it (QEMU_SPLASH=1; use
# tools/splash-test.sh, which adds the sound device and checks the
# screenshots and the capture): the dark screen before init, two frames of
# the animation a second apart, the shell's text after the hand-back, then
# the splash's self-test from the shell.
wait 120 init: hello from ring 3
shot splash-quiet
wait 120 splash: first frame
sleep 1
shot splash-a
sleep 1
shot splash-b
wait 60 console: the screen is back
seen 60 jam>
shot splash-shell
send run splash --selftest
wait 240 splash: selftest
wait 240 run: bin/splash exited
wait jam>
send reboot
wait reboot: resetting
