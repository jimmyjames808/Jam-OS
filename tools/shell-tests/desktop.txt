# The desktop with real input (tools/mouse-test.sh's second boot, the
# compositor's: a plain boot with a usb-kbd on xhci port 2 and a usb-mouse
# on port 3): QEMU's keys and mouse go usb-bus -> drv/hid -> the compositor
# (devmgr's INPUT channel) -> the focused window, the terminal's console
# -> the shell. The windows tile (the default): typing in the first
# terminal; Super+Enter opening terminal 2 beside it, whose window takes
# the keys; a click on the first terminal giving them back; Super+Right
# and Super+H moving the focus by direction; two more terminals splitting
# the focused tile (four tiles: desktop-tiles); the gap between the first
# two dragged; Super+Q closing terminals; Super+T's floating saved and kept
# across a reboot.
#
# Where things are at OVMF's 1280x800: the room for tiles is under the
# desktop's strip (40 pixels) less a 6-pixel gap all round: x 6..1274, y
# 46..794. The first terminal fills it; terminal 2 halves it across (the
# first x 6..637, terminal 2 x 643..1274, the gap between at x 637..643);
# terminal 3, with the first focused, halves the first's tile top and
# bottom (y 46..417 and 423..794); terminal 4 halves terminal 3's across.
# `pointer` puts the mouse there exactly (serial-feed.py: 1:1 steps).
wait 120 Jam OS shell
wait jam>
seen 30 report mouse ready
seen 30 compositor: keyboard 0627:0001 ready
usbkeys echo usb-one\r
wait 10 usb-one
wait jam>
# Super+Enter, whichever window has the keys: terminal 2, the right half
monitor sendkey meta_l-ret
wait 10 init: terminal 2 opens
wait 30 Jam OS shell
wait 10 jam>
sleep 1
usbkeys echo usb-two\r
wait 10 usb-two
wait jam>
shot desktop-two
# a click on the first terminal (the left half): the keys go back to it
pointer 300 400
monitor mouse_button 1
sleep 0.2
monitor mouse_button 0
sleep 0.5
usbkeys echo usb-back\r
wait 10 usb-back
wait jam>
# Super+Right: the focus to the window on the right (terminal 2) ...
monitor sendkey meta_l-right
sleep 0.5
usbkeys echo usb-right\r
wait 10 usb-right
wait jam>
# ... and Super+H back to the left (the first)
monitor sendkey meta_l-h
sleep 0.5
usbkeys echo usb-left\r
wait 10 usb-left
wait jam>
# two more: terminal 3 splits the focused first's tile (top and bottom),
# terminal 4 splits terminal 3's (side by side)
monitor sendkey meta_l-ret
wait 10 init: terminal 3 opens
wait 30 Jam OS shell
wait 10 jam>
sleep 1
monitor sendkey meta_l-ret
wait 10 init: terminal 4 opens
wait 30 Jam OS shell
wait 10 jam>
sleep 1
usbkeys echo usb-four\r
wait 10 usb-four
wait jam>
sleep 1
shot desktop-tiles
# the gap between the first and terminal 2 (x 637..643) dragged 100 right
pointer 640 300
monitor mouse_button 1
sleep 0.2
pointer 740 300
sleep 0.2
monitor mouse_button 0
sleep 1
shot desktop-resized
# Super+Q closes the focused window (terminal 4), then terminal 3 (the
# focus moved to it): its console ends, init closes the terminal
monitor sendkey meta_l-q
wait 10 init: terminal 4: its window was closed
wait 10 init: terminal 4 is closed
sleep 1
pointer 300 600
monitor mouse_button 1
sleep 0.2
monitor mouse_button 0
sleep 0.5
monitor sendkey meta_l-q
wait 10 init: terminal 3: its window was closed
wait 10 init: terminal 3 is closed
sleep 1
shot desktop-closed
usbkeys echo usb-last\r
wait 10 usb-last
wait jam>
# Super+T: floating, saved by init (the compositor's layout_wait answered);
# a reboot (kexec) comes back floating, as the settings say
monitor sendkey meta_l-t
wait 10 init: the window layout is floating (saved)
send grep layout /data/etc/settings
wait 10 display.layout = floating
wait jam>
shot desktop-floating
send reboot
wait 60 kexec: started by a reboot
wait 120 init: the window layout: floating, as the settings say
seen 120 init: the shell is up
sleep 2
send reboot -f
wait reboot: resetting
