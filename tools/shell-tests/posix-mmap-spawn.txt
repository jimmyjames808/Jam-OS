# posix_spawn of a program on /data (tools/posix-mmap-test.sh, case spawn):
# a copy of posixtest at /data/pt-prog is EACCES while it isn't allowed;
# then `allow` (y to its question) and the same spawn again: init's
# /svc/exec seals and starts it, and it exits 7; started so again, and from
# the shell, its self-image section passes (its /proc/self/exe is its path,
# and a spawn of /proc/self/exe runs from its own image: docs/CR-PLAN.md,
# "CL2 notes"); a forked child's execve of it (spawn, wait, exit) passes our
# pipe on and forwards our SIGTERM to it (exit 9: CR-PLAN, "CCS"), and our
# SIGKILL to such a child ends the program too (FORK-PLAN, "FJ1 notes"); and
# `posixtest spawn-wmap` (WRITEMAP-PLAN's WT1): with a writable shared
# mapping of the file alive (an O_RDWR descriptor's, open and then closed)
# the file is ETXTBSY to a spawn and to a forked child's execve, and runs
# again once the mapping is gone.
# QEMU_INPUT=this file, command line "shell".
wait 120 Jam OS shell
wait {prompt}
seen 60 init: /data mounted
send T=s
wait {prompt}
send cp /boot/bin/posixtest /data/pt-prog; echo cp-$T=$?
wait 60 cp-s=0
wait {prompt}
send posixtest spawn-data /data/pt-prog refused; echo r-$T=$?
wait 60 posixtest: spawn-data: ok
wait 30 r-s=0
wait {prompt}
send allow /data/pt-prog; echo al-$T=$?
wait 60 ? y/n
type y
wait 60 al-s=
wait {prompt}
send posixtest spawn-data /data/pt-prog allowed; echo a-$T=$?
wait 60 posixtest: spawn-data: /data/pt-prog:
wait 60 posixtest: spawn-data: /data/pt-prog self-image: PASS
wait 60 posixtest: spawn-data: /data/pt-prog fork-exec: wait status 0x900
wait 60 posixtest: spawn-data: /data/pt-prog fork-kill: wait status 0x9, the program gone
wait 30 a-s=0
wait {prompt}
send run /data/pt-prog self-image /data/pt-prog; echo si-$T=$?
wait 60 posixtest: self-image: ok
wait 30 si-s=0
wait {prompt}
send posixtest spawn-wmap /data/pt-prog; echo wm-$T=$?
wait 90 posixtest: spawn-wmap: ok
wait 30 wm-s=0
wait {prompt}
send reboot -f
wait reboot: resetting
