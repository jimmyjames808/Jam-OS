# posix_spawn of a program on /data (tools/posix-mmap-test.sh, case spawn):
# a copy of posixtest at /data/pt-prog is EACCES while it isn't allowed;
# then `allow` (y to its question) and the same spawn again, whose line
# the script shows (it runs only once something seals a program for a
# POSIX program; until then it is EACCES and that run's section fails,
# which the script doesn't count).
# QEMU_INPUT=this file, command line "shell".
wait 120 Jam OS shell
wait {prompt}
seen 60 init: /data mounted
send T=s
wait {prompt}
send cp /boot/bin/posixtest /data/pt-prog; echo cp-$T=$?
wait 60 cp-s=0
wait {prompt}
send posixtest spawn-data /data/pt-prog refused; echo r-$T=$?
wait 60 posixtest: spawn-data: ok
wait 30 r-s=0
wait {prompt}
send allow /data/pt-prog; echo al-$T=$?
wait 60 ? y/n
type y
wait 60 al-s=
wait {prompt}
send posixtest spawn-data /data/pt-prog allowed; echo a-$T=$?
wait 60 posixtest: spawn-data: /data/pt-prog:
wait 30 a-s=
wait {prompt}
send reboot -f
wait reboot: resetting
