# `pkg` and a program as big as Chromium's, the second boot (tools/pkg-test.sh,
# case big): the stick and the Jam OS drive from cold. The package installed
# by pkg-big.txt is still there, its approval (the stick's) and its files
# (/jam's) agree, it runs, and `pkg remove` takes its 300 MiB and its
# approval away. The test script puts the program's hash where @HASH@ and its
# page count where @PAGES@ are.
wait 120 Jam OS shell
seen 60 init: /data mounted
seen 90 init: /jam: mounted as views
seen 60 netstack: link up
seen 30 netstack: address 10.2.21.5/24
wait {prompt}
send T=x
wait {prompt}
send pkg list
wait bigpkg 1.0-r1
wait {prompt}
send allow -l
seen 5 [package bigpkg 1.0-r1]
wait {prompt}
send run /jam/pkg/bin/bigpkg
wait 300 bigpkg: @PAGES@ pages, every 16: hash @HASH@
wait {prompt}
send pkg remove bigpkg
wait Remove 1 package? [y/N]
type y
wait Removed bigpkg
wait {prompt}
send pkg list
wait No package is installed.
wait {prompt}
send allow -l
wait no program on /data is allowed to run
wait {prompt}
send ls /jam/pkg/bin | grep -c bigpkg | grep ^0$ && echo ok-$T-gone
wait ok-x-gone
wait {prompt}
send sync
wait {prompt}
send reboot -f
wait reboot: resetting
