# Records stored through writable shared mappings on /jam, an ext volume
# served by extfs with the mount's pager (docs/WRITEMAP-PLAN.md's WT1;
# tools/pager-test.sh's wmsync case, which makes the drive, then checks this
# boot's log and the disk it leaves, then boots pager-wmsync2.txt over it).
# `pagertest rec` stores 200 records of 5000 bytes (rec.c's header) through
# a mapping of the file, in three runs:
#   A  msynced (every 10 records; the line "acked 200" says all are safe),
#      the writer holding its mapping while `kill extfs-jam` kills extfs:
#      the successor serves the file: every record whole by read and by a
#      new mapping, and by the writer's own mapping after.
#   B  never msynced, the writer holding its mapping while extfs is killed
#      with all 200 stores dirty: the successor has them (read and a new
#      mapping show every record) and writes them by age (files.writeback:
#      10 s; the script sleeps 16).
#   C  msynced and then QEMU quit at once (a power cut with no stop of the
#      volume): the next boot finds the records, and so does Linux on a copy
#      of the disk (the case's host side).
# As in cmds.txt, a marker built from $T (m) never appears in the typed
# line, only when the command before it ran.
wait 120 Jam OS shell
wait {prompt}
seen 60 init: /data mounted
send T=m
wait {prompt}
seen 60 init: /jam mounted

# ---- A: msynced, extfs killed while the writer holds its mapping -------------------------------
send pagertest rec make /jam/ra.bin 200 5000; echo a0-$T=$?
wait 60 a0-m=0
wait {prompt}
send pagertest rec put /jam/ra.bin 200 11 msync 10 5000 /jam/cueA &
wait 60 pagertest: rec: holding, waiting for /jam/cueA
seen 10 pagertest: rec: acked 200
send kill extfs-jam; echo ka-$T=$?
wait ka-m=0
wait {prompt}
send pagertest rec check /jam/ra.bin 200 11 200 5000; echo a1-$T=$?
wait 120 pagertest: rec: check /jam/ra.bin: PASS
wait a1-m=0
wait {prompt}
send write /jam/cueA go; echo a2-$T=$?
wait a2-m=0
seen 60 pagertest: rec: held mapping of /jam/ra.bin: 200 whole of 200: PASS
wait {prompt}

# ---- B: never msynced, extfs killed with every store dirty --------------------------------------
send pagertest rec make /jam/rb.bin 200 5000; echo b0-$T=$?
wait 60 b0-m=0
wait {prompt}
send pagertest rec put /jam/rb.bin 200 12 nosync 1 5000 /jam/cueB &
wait 60 pagertest: rec: holding, waiting for /jam/cueB
send kill extfs-jam; echo kb-$T=$?
wait kb-m=0
wait {prompt}
send pagertest rec check /jam/rb.bin 200 12 200 5000; echo b1-$T=$?
wait 120 pagertest: rec: check /jam/rb.bin: PASS
wait b1-m=0
wait {prompt}
send write /jam/cueB go; echo b2-$T=$?
wait b2-m=0
seen 60 pagertest: rec: held mapping of /jam/rb.bin: 200 whole of 200: PASS
wait {prompt}
# (the writer is gone; its stores are written back by age, from the VMO the kernel kept)
send sleep 16; echo b3-$T=$?
wait 60 b3-m=0
wait {prompt}

# ---- C: msynced, then the power cut ----------------------------------------------------------------
send pagertest rec make /jam/rc.bin 200 5000; echo c0-$T=$?
wait 60 c0-m=0
wait {prompt}
send pagertest rec put /jam/rc.bin 200 13 msync 10 5000; echo c1-$T=$?
wait 120 pagertest: rec: put /jam/rc.bin: done
wait c1-m=0
wait {prompt}
monitor quit
