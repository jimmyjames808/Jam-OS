# The mixer without a warm spare (the boot word `nospare`); use
# tools/mixer-spare-test.sh. No spare is started; `kill mixer` starts a
# new process (said with the time it took and the kill to the first
# answer), which answers.
# (The markers are echoed as np-""1 so the typed line itself never matches.)
wait 120 Jam OS shell
wait {prompt}
seen 60 init: nospare: no warm spare
send vol; echo np-""1
wait master 0.0 dB
wait np-1
wait {prompt}
send kill mixer
wait init: bin/mixer: killed on purpose
wait init: bin/mixer: new process started (no spare waited)
wait init: bin/mixer: kill to first answer
send kill mixer
wait init: bin/mixer: kill to first answer
send kill mixer
wait init: bin/mixer: kill to first answer
send kill mixer
wait init: bin/mixer: kill to first answer
send kill mixer
wait init: bin/mixer: kill to first answer
send vol; echo np-""3
wait master 0.0 dB
wait np-3
wait {prompt}
send ps; echo np-""4
wait np-4
wait {prompt}
send reboot -f
