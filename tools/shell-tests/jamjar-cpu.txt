# jamjar's CPU use while it plays in a big window (tools/jamjar-test.sh with
# JAMJAR_CPU=1: the PC's 2560x1440, the same library and player as
# jamjar.txt). jamjar runs in the background and takes the whole screen
# (Super+F); its keys come from QEMU's keyboard (monitor sendkey) so that
# the serial shell stays free for `top`, which prints each process's CPU
# use over the last 10 s (100.0 is one CPU). The test script reads jamjar's
# row (and the compositor's, which shows what the window costs it) from the
# second frame, and jamjar's own frame counts when it quits.
# JAMJAR_ARGS adds words to the jamjar command (`whole`: the old way,
# redrawing the whole window every frame).
wait 120 Jam OS shell
wait {prompt}
seen 60 init: /data mounted
seen 60 music: ready
send jamjar trace &
wait 30 jamjar: in the background
wait 30 jamjar: library /data/music: 6 artists, 7 albums, 9 tracks
monitor sendkey meta_l-f
wait 30 jamjar: resized to 2560x1440
sleep 1
monitor sendkey a
wait 20 jamjar: play everything
wait 30 jamjar: hearing
wait 20 jamjar: spectrum: loudest bar
sleep 4
send top -n 2 -d 10 | grep -E "top - up|jamjar|compositor"; echo cpu-""1 $?
wait 60 cpu-1 0
wait {prompt}
monitor sendkey q
wait 30 the music plays on
wait 10 frames redrew only
send music stop
wait music: stopped
wait {prompt}
send reboot -f
wait reboot: resetting
