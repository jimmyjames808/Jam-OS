# Review: a command line longer than the screen row. The console's current
# line is one row: an echo past its end wraps (commits a row), and a redraw
# ("\r" + prompt + the whole line) would commit another copy of the first
# row each time. The shell keeps the line on its row, scrolled sideways
# (user/services/shell/main.c): the shot shows one row, '<' at its left; at
# Enter the whole line is written once, wrapped, and runs (the shell says
# the 186-character word, 180 a's, an X and 5 b's, is an unknown command).
wait 120 Jam OS shell
wait {prompt}
sleep 1
send clear
wait {prompt}
type aaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaabbbbbbbbbb
type \x7f\x7f\x7f\x7f\x7f
type \e[D\e[D\e[D\e[D\e[D
type X
sleep 1
shot longline
send
wait aaaaaXbbbbb: unknown command
sleep 0.5
shot longline-after
send reboot -f
wait reboot: resetting
