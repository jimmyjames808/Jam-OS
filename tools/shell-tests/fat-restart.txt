# fat carries on from a dead instance's state (user/services/fat/adopt.c);
# use tools/fat-restart-test.sh (it checks the log and the stick afterwards).
# On the boot stick's /data, through devmgr: `kill fat-data` right after a
# write that was never synced (and right after a file was closed, which
# fat finishes after its client has gone): the successor carries on, the
# data is there; a copy of the kernel then a kill, its checksum the
# kernel's; bursts of kills between appends to one file; the mount never
# goes and nothing is given up on. Booted once with the warm spare and once
# with `nospare` (the same script: a new process adopts the state just
# the same).
# devmgr's and fat's lines reach the serial port through the kernel log,
# the shell's through the console: their order isn't fixed, hence `seen`.
# (The markers are echoed as fr-""1 so the typed line itself never matches.)
wait 120 Jam OS shell
wait {prompt}
seen 60 init: /data mounted
send mkdir /data/fr; echo fr-""1
wait fr-1
wait {prompt}

# ---- unsynced writes, killed right after -----------------------------------------------------
send sleep 1; write /data/fr/u1.txt unsynced-""1 && kill fat-data && cat /data/fr/u1.txt; echo fr-""2
wait unsynced-1
wait fr-2
send sleep 1; write /data/fr/u2.txt unsynced-""2 && kill fat-data && cat /data/fr/u2.txt; echo fr-""3
wait unsynced-2
wait fr-3
send sleep 1; write /data/fr/u3.txt unsynced-""3 && kill fat-data && cat /data/fr/u3.txt; echo fr-""4
wait unsynced-3
wait fr-4
send write -a /data/fr/u1.txt again-""1 && kill fat-data && kill fat-data && cat /data/fr/u1.txt; echo fr-""5
wait again-1
wait fr-5
wait {prompt}
seen 20 fat /data: restart (killed

# ---- a copy, then kills; its checksum ----------------------------------------------------------
send sleep 1; cp /esp/boot/jamos.elf /data/fr/k1.bin && kill fat-data && sha256sum /data/fr/k1.bin /esp/boot/jamos.elf; echo fr-""6
wait fr-6
wait {prompt}
send repeat 3 kill fat-data; cp /esp/boot/bootfs.img /data/fr/k2.bin && repeat 2 kill fat-data; sha256sum /data/fr/k2.bin /esp/boot/bootfs.img; echo fr-""7
wait fr-7
wait {prompt}

# ---- bursts of kills between appends to one file -----------------------------------------------
# (short lines: the shell cuts a line that is too long)
send repeat 5 kill fat-data; write -a /data/fr/burst.txt b-""1; echo fr-""8
wait fr-8
wait {prompt}
send repeat 5 kill fat-data; write -a /data/fr/burst.txt b-""2; repeat 5 kill fat-data; echo fr-""9
wait fr-9
wait {prompt}
send cat /data/fr/burst.txt; ls /data/fr; echo fr-""11
wait b-1
wait b-2
wait fr-11
wait {prompt}
send sync; ps; echo fr-""10
wait fr-10
wait {prompt}
send reboot -f
wait reboot: resetting
