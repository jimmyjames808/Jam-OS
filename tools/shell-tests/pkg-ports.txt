# The libarchive chain through `pkg` (tools/pkg-ports-test.sh; docs/M13.7-PLAN.md, PO9):
# real packages (zlib, bzip2, xz, libarchive, built by tools/ports.py) in the repository
# `real` of tools/pkg-test-repo.py, joined by the network peer's TCP relay (the guest
# connects to 10.2.21.174:8000). The image's /boot/ports.pub is the ports vectors'
# public test key, and so is the repository's. The repository `conflict` is used last: its
# serial is one above `real`'s, so `real` would be an old index after it.
wait 120 Jam OS shell
seen 60 init: /data mounted
seen 60 netstack: link up
seen 30 netstack: address 10.2.21.5/24
wait {prompt}
send T=x
wait {prompt}
send P=/data/pkg/bin
wait {prompt}
send S=/boot/usr/bin
wait {prompt}

# ---- the repository has the four packages (a -test subpackage is never in it) -----------
send pkg update
wait The repository has 4 packages (serial
wait Everything installed is up to date.
wait {prompt}

# ---- one question for libarchive and what it runs; the three go in first, in order ------
send pkg install libarchive
wait Install 4 packages, 12 programs? [y/N]
type y
wait Installed bzip2 1.0.8-r1
wait Installed xz 5.8.4-r1
wait Installed zlib 1.3.2-r1
wait Installed libarchive 3.8.10-r1
wait {prompt}
send pkg list
wait bzip2 1.0.8-r1
wait libarchive 3.8.10-r1
wait xz 5.8.4-r1
wait zlib 1.3.2-r1
wait {prompt}
send pkg check
wait pkg check: 4 packages, every file is as its manifest says
wait {prompt}
send pkg info libarchive
wait libarchive 3.8.10-r1 (installed)
wait runs: bzip2
wait runs: xz
wait runs: zlib
wait {prompt}
send allow -l | grep -c package | run $S/sed s/^/n-/
wait n-12
wait {prompt}
send ls /data/pkg/bin | grep -c ^example$ | run $S/sed s/^/n-/
wait n-0
wait {prompt}
send ls /data/pkg/share | grep -c ^bzip2 | run $S/sed s/^/n-/
wait n-0
wait {prompt}

# ---- bsdtar in a round trip through each of the three, read back by the other programs --
send mkdir /data/pk-t
wait {prompt}
send cp /data/pk-in /data/pk-t/a.txt
wait {prompt}
send run $P/bsdtar -cjf /data/pk.tbz -C /data pk-t; echo st-$T=$?
wait st-x=0
wait {prompt}
send run $P/bzip2 -t /data/pk.tbz; echo st-$T=$?
wait st-x=0
wait {prompt}
send run $P/bsdtar -cJf /data/pk.txz -C /data pk-t; echo st-$T=$?
wait st-x=0
wait {prompt}
send run $P/xz -t /data/pk.txz; echo st-$T=$?
wait st-x=0
wait {prompt}
send run $P/bsdtar -czf /data/pk.tgz -C /data pk-t; echo st-$T=$?
wait st-x=0
wait {prompt}
send run $P/minigzip -d -c /data/pk.tgz | run $P/bsdtar -tf - | run $S/grep -c ^pk-t | run $S/sed s/^/n-/
wait n-2
wait {prompt}
send mkdir /data/pk-o; run $P/bsdtar -xf /data/pk.txz -C /data/pk-o; echo st-$T=$?
wait st-x=0
wait {prompt}
send cmp /data/pk-t/a.txt /data/pk-o/pk-t/a.txt && echo ok-$T-tree
wait ok-x-tree
wait {prompt}
