# The pager's plumbing (tools/pager-test.sh, case plumb): what volumes and
# libos do for mapped files, and fat's answers to file.map. Boot stick a
# (the script makes it) comes in as /usb0 and goes again.
#   pagertest self    libos's mapped-file code over VMOs that are no pager's
#   pagertest map     file.map to fat: a VMO, from /data, /esp and the
#                     stick, also after fat-data and fat-usb0 were killed
#                     (the request goes through fat's kept-channel
#                     machinery); a directory and a missing file refused
#   pagertest hash, touch   a small file's pages through the mapping, by
#                     loads and by a system call's copy
#   the pagertest commands' arguments are checked
# volumes' pager lines (one pager a mount, kept across fat-data's restart,
# closed with the stick) are checked by the script from the log.
# QEMU_INPUT=this file, command line "shell", the drive aimg. As in
# cmds.txt, the text waited for is built from $T, so it never appears in
# the typed line.
wait 120 Jam OS shell
wait {prompt}
seen 60 init: /data mounted
send T=p
wait {prompt}

# ---- libos's mapping code, no mount needed ----------------------------------------------------
send pagertest self; echo self-$T=$?
wait pagertest: self: PASS
wait self-p=0
wait {prompt}

# ---- fat's answers -------------------------------------------------------------------------------
send write /data/pager.txt a file to map; echo w-$T=$?
wait w-p=0
wait {prompt}
send pagertest map /data/pager.txt; echo m1-$T=$?
wait pagertest: map /data/pager.txt: OK
wait m1-p=0
wait {prompt}
send pagertest map /esp/boot/jamos.elf; echo m2-$T=$?
wait pagertest: map /esp/boot/jamos.elf: OK
wait m2-p=0
wait {prompt}
send pagertest map /data; echo m3-$T=$?
wait pagertest: map /data: ERR_IS_DIR
wait m3-p=2
wait {prompt}
send pagertest map /data/none.txt; echo m4-$T=$?
wait pagertest: map /data/none.txt: ERR_NOT_FOUND
wait m4-p=2
wait {prompt}
send pagertest hash /data/pager.txt; echo h1-$T=$?
wait /data/pager.txt: 14 bytes: mapped sha256
wait h1-p=0
wait {prompt}
send pagertest touch /data/pager.txt 2 load; echo t1-$T=$?
wait pagertest: touched /data/pager.txt at 2: byte 66
wait t1-p=0
wait {prompt}
send pagertest touch /data/pager.txt 12 copy; echo t2-$T=$?
wait pagertest: touch /data/pager.txt at 12 by a copy: OK
wait t2-p=0
wait {prompt}

# ---- the commands' own arguments ------------------------------------------------------------------
send pagertest; echo u1-$T=$?
wait usage: pagertest self
wait u1-p=64
wait {prompt}
send pagertest touch /data/pager.txt 0 jump; echo u2-$T=$?
wait u2-p=64
wait {prompt}
send pagertest read /data/pager.txt 0 0 0; echo u3-$T=$?
wait u3-p=64
wait {prompt}

# ---- a stick in: its own pager; fat-data killed: the same pager ---------------------------------
monitor device_add usb-storage,id=sa,bus=xhci.0,port=2,drive=aimg
wait 60 init: /usb0 mounted
send pagertest map /usb0/pager.bin; echo m5-$T=$?
wait pagertest: map /usb0/pager.bin: OK (2097152 bytes)
wait m5-p=0
wait {prompt}
send sleep 1; kill fat-data; echo k-$T=$?
wait shell: killed process
wait k-p=0
wait {prompt}
seen 10 volumes: fat-data: spare promoted
send pagertest map /data/pager.txt; echo m6-$T=$?
wait pagertest: map /data/pager.txt: OK (14 bytes)
wait m6-p=0
wait {prompt}
send kill fat-usb0; echo k2-$T=$?
wait k2-p=0
wait {prompt}
send pagertest map /usb0/pager.bin; echo m7-$T=$?
wait pagertest: map /usb0/pager.bin: OK (2097152 bytes)
wait m7-p=0
wait {prompt}

# ---- the stick pulled: its mount and its pager go ----------------------------------------------
monitor device_del sa
wait 30 init: /usb0 is gone
send pagertest map /usb0/pager.bin; echo m8-$T=$?
wait pagertest: map /usb0/pager.bin: ERR_NOT_FOUND
wait m8-p=2
wait {prompt}
send reboot -f
wait reboot: resetting
